## 10. Multiply two matrices

The number of columns in the first matrix must equal the number of rows in the second.

```cpp
#include iostream
using namespace std;

int main() {
    int r1, c1, r2, c2;
    int a[10][10], b[10][10], product[10][10] = {};

    cout  Enter rows and columns of first matrix ;
    cin  r1  c1;

    cout  Enter rows and columns of second matrix ;
    cin  r2  c2;

    if (c1 != r2) {
        cout  Matrix multiplication is not possible.;
        return 0;
    }

    cout  Enter first matrixn;
    for (int i = 0; i  r1; i++)
        for (int j = 0; j  c1; j++)
            cin  a[i][j];

    cout  Enter second matrixn;
    for (int i = 0; i  r2; i++)
        for (int j = 0; j  c2; j++)
            cin  b[i][j];

    for (int i = 0; i  r1; i++)
        for (int j = 0; j  c2; j++)
            for (int k = 0; k  c1; k++)
                product[i][j] += a[i][k]  b[k][j];

    cout  Product matrixn;
    for (int i = 0; i  r1; i++) {
        for (int j = 0; j  c2; j++)
            cout  product[i][j]   ;
        cout  n;
    }

    return 0;
}
```

## 11. Sort an array using Bubble Sort in ascending order

```cpp
#include iostream
using namespace std;

int main() {
    int n, a[100];

    cout  Enter number of elements ;
    cin  n;

    cout  Enter   n   elementsn;
    for (int i = 0; i  n; i++)
        cin  a[i];

    for (int i = 0; i  n - 1; i++)
        for (int j = 0; j  n - 1 - i; j++)
            if (a[j]  a[j + 1]) {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }

    cout  Ascending order ;
    for (int i = 0; i  n; i++)
        cout  a[i]   ;

    return 0;
}
```

## 15. Search for an element using Binary Search

Enter the array in ascending order before searching.

```cpp
#include iostream
using namespace std;

int main() {
    int n, a[100], target;
    int low, high, mid, position = -1;

    cout  Enter number of elements ;
    cin  n;

    cout  Enter   n   elements in ascending ordern;
    for (int i = 0; i  n; i++)
        cin  a[i];

    cout  Enter element to search for ;
    cin  target;

    low = 0;
    high = n - 1;

    while (low = high) {
        mid = low + (high - low)  2;

        if (a[mid] == target) {
            position = mid;
            break;
        } else if (a[mid]  target) {
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }

    if (position == -1)
        cout  Element not found;
    else
        cout  Element found at position   position + 1;

    return 0;
}
```

## 20. Search for the largest element in a sorted array using Binary Search

For an ascending array, the largest element is at the final position. Binary Search can locate it by finding the last array index.

```cpp

using namespace std;

int main() {
    int n, a[100];
    int low, high, mid, lastIndex = -1;

    cout  Enter number of elements ;
    cin  n;

    cout  Enter elements in ascending ordern;
    for (int i = 0; i  n; i++)
        cin  a[i];

    low = 0;
    high = n - 1;

     Find the last valid index using Binary Search.
    while (low = high) {
        mid = low + (high - low)  2;

        if (mid == n - 1  a[mid] = a[n - 1]) {
            lastIndex = mid;
            low = mid + 1;
        } else {
            high = mid - 1;
        }
    }

    if (lastIndex == -1)
        cout  Array is empty.;
    else
        cout  Largest element =   a[lastIndex];

    return 0;
}
```
